```r
.tmle_args <- function(Y, A, X, g, Q1, Q0, truncation, which) {
  # tmle refits a treatment model for the ATT when the overlap subset drops more
  # than a tenth of the controls, whatever g1W it was given. Its default library
  # (glm, dbarts, gam with 10 folds) takes minutes and is not cross-fitted, so
  # the refit is limited to a logistic regression.
  args <- list(Y = Y, A = A, W = as.data.frame(X), Q = cbind(Q0, Q1), g1W = g,
               gbound = c(truncation, 1 - truncation), family = "binomial",
               g.SL.library = "SL.glm")
  # tmle also runs its ATT and ATC targeting on every call unless told not to;
  # in a profile of one simulation repetition that took about 70% of the time.
  # Only the ATT reads it.
  if (which != "ATT" && "evalATT" %in% names(formals(tmle::tmle))) args$evalATT <- FALSE
  args
}

.target_tmle <- function(Y, A, X, g, Q1, Q0, truncation, which) {
  if (length(unique(A)) < 2L)
    stop("Only one treatment arm in the target population.", call. = FALSE)
  f <- suppressMessages(suppressWarnings(do.call(
    tmle::tmle, .tmle_args(Y, A, X, g, Q1, Q0, truncation, which))))
  e <- f$estimates[[which]]
  r1 <- if (which == "ATE") f$estimates$EY1$psi else mean(Y[A == 1])
  r0 <- if (which == "ATE") f$estimates$EY0$psi else r1 - e$psi
  list(estimate = e$psi, se = sqrt(e$var.psi), ci_lower = e$CI[1], ci_upper = e$CI[2],
       n_population = if (which == "ATT") sum(A == 1) else length(Y),
       risk1 = r1, risk0 = r0)
}
```

```r
# Smallest group whose covariate means the dossier may report.
.min_cell <- 5L

.std_shift <- function(m, ref, s) ifelse(is.finite(s) & s > 0, (m - ref) / s, 0)

# Covariate means in each estimand's target population, and the standardized
# shift from the full cohort (difference over the full-cohort SD).
.populations <- function(X, A, g, estimands, band) {
  mu <- colMeans(X)
  s <- apply(X, 2, stats::sd)
  out <- data.frame(covariate = colnames(X), cohort = unname(mu), stringsAsFactors = FALSE)
  for (e in estimands) {
    w <- switch(e,
      ATE = rep(1, nrow(X)),
      trimmed_ATE = as.numeric(g >= band[1] & g <= band[2]),
      ATT = as.numeric(A == 1),
      ATO = g * (1 - g))
    m <- if (sum(w) > 0) colSums(X * w) / sum(w) else rep(NA_real_, ncol(X))
    out[[e]] <- unname(m)
    out[[paste0(e, "_shift")]] <- unname(.std_shift(m, mu, s))
  }
  out
}

# The patients a trimmed analysis no longer describes: counts by arm, and the
# covariate means of removed and kept patients. Means of fewer than .min_cell
# removed patients are suppressed.
.trimmed_profile <- function(X, A, g, band) {
  out <- g < band[1] | g > band[2]
  s <- apply(X, 2, stats::sd)
  mean_or_na <- function(rows) if (sum(rows) >= .min_cell) colMeans(X[rows, , drop = FALSE]) else
    rep(NA_real_, ncol(X))
  removed <- mean_or_na(out)
  kept <- mean_or_na(!out)
  list(
    counts = data.frame(group = c("all", "treated", "control"),
                        removed = c(sum(out), sum(out & A == 1), sum(out & A == 0)),
                        kept = c(sum(!out), sum(!out & A == 1), sum(!out & A == 0)),
                        stringsAsFactors = FALSE),
    covariates = data.frame(covariate = colnames(X), removed = unname(removed),
                            kept = unname(kept),
                            std_diff = unname(ifelse(is.na(removed), NA_real_,
                                                     .std_shift(removed, kept, s))),
                            stringsAsFactors = FALSE))
}
```

```r
# Area under the ROC curve of g for A, by the rank (Mann-Whitney) formula.
.c_statistic <- function(g, A) {
  r <- rank(g)
  n1 <- sum(A == 1)
  n0 <- sum(A == 0)
  (sum(r[A == 1]) - n1 * (n1 + 1) / 2) / (n1 * n0)
}
```

```r
#' Export the design log
#'
#' Assembles the log from the objects it receives: plan, hashes, design
#' summaries, decision, approval, overrides and estimates. Nothing is
#' stored between calls.
#'
#' @param x A [unblind()] result, or a [design_report()] dossier for the
#'   pre-outcome log.
#' @param fit Optional `cr_estimate` or list of them.
#' @param file Optional path: `.json` for `format = "json"`, `.csv` for
#'   `format = "muntner"`.
#' @param format `"json"` (the full log) or `"muntner"` (the Table S1
#'   columns of Muntner et al. 2024: date, stage, issue, decision,
#'   rationale, decided_by).
#' @return The log (a list for JSON, a data frame for Muntner), invisibly
#'   when `file` is given.
#' @references Muntner P et al. (2024) Pharmacoepidemiol Drug Saf
#'   33:e5770, supplementary Table S1.
#' @export
export_design_log <- function(x, fit = NULL, file = NULL, format = c("json", "muntner")) {
  format <- match.arg(format)
  if (inherits(x, "cr_unblinded")) {
    .verify_unblinded(x)
    d <- x$dossier
    approval <- x$approval
  } else if (inherits(x, "cr_dossier")) {
    d <- x
    approval <- NULL
  } else {
    stop("`x` must be an unblind() result or a design_report() dossier.", call. = FALSE)
  }
  .verify_dossier(d)
  fits <- if (inherits(fit, "cr_estimate")) list(fit) else fit
  if (format == "muntner") {
    out <- .muntner_log(d, approval, fits)
    if (is.null(file)) return(out)
    utils::write.csv(out, file, row.names = FALSE)
    return(invisible(out))
  }
  log <- list(
    package_version = d$version, lock_hash = d$lock_hash, dossier_hash = d$dossier_hash,
    lock_created = d$lock_created, dossier_created = d$created, plan = d$plan,
    decision = d$decision,
    design = d$design[intersect(c("overlap", "overlap_grade", "share_outside_band",
                                  "c_statistic", "ess", "balance", "edge", "populations",
                                  "trimmed_profile"), names(d$design))],
    simulation = d$simulation[c("verdict", "scope", "reps", "K")],
    expected_events = d$expected_events_range,
    negative_controls = d$nc[c("by_rung", "verdict")],
    notes = d$notes, approval = approval,
    estimates = lapply(fits, function(f) unclass(f)))
  if (is.null(file)) return(log)
  jsonlite::write_json(log, file, auto_unbox = TRUE, pretty = TRUE, digits = NA,
                       null = "null", na = "null")
  invisible(log)
}
```

```r
# Builds cr_example: design data (no outcome), outcomes, and the true value of
# each estimand, joined by id. `assign` is the log-odds of treatment per SD of
# severity; `modifier` is how much the effect on the log-odds scale grows per
# SD of severity. Both were chosen so that, under the vignette's plan, the ATE
# is infeasible while a lower rung of the ladder is feasible.
make_cr_example <- function(assign, modifier, seed = 20261005, n = 1500) {
  set.seed(seed)
  age <- round(stats::rnorm(n, 60, 12))
  female <- stats::rbinom(n, 1, 0.5)
  severity <- stats::rnorm(n)
  prior_visits <- stats::rpois(n, 2)
  transfer <- stats::rbinom(n, 1, 0.1)
  ps <- stats::plogis(-0.3 + assign * severity + 0.02 * (age - 60) - 0.3 * female)
  A <- stats::rbinom(n, 1, ps)
  health_seeking <- stats::rbinom(n, 1, stats::plogis(-0.7 + 0.3 * prior_visits))
  risk <- function(a) stats::plogis(-2 + (0.4 + modifier * severity) * a + 0.8 * severity +
                                      0.03 * (age - 60) + 0.3 * severity^2)
  death_30d <- stats::rbinom(n, 1, risk(A))
  tau <- risk(1) - risk(0)
  inb <- ps >= 0.05 & ps <= 0.95
  h <- ps * (1 - ps)
  list(
    design = data.frame(id = seq_len(n), A = A, age = age, female = female,
                        severity = severity, prior_visits = prior_visits,
                        transfer = transfer, health_seeking = health_seeking),
    outcomes = data.frame(id = seq_len(n), death_30d = death_30d),
    truth = data.frame(
      estimand = c("ATE", "trimmed_ATE", "ATT", "ATO"),
      truth = c(mean(tau), mean(tau[inb]), sum(ps * tau) / sum(ps), sum(h * tau) / sum(h)),
      population = c("all patients",
                     "patients whose true propensity score lies in [0.05, 0.95]",
                     "patients weighted by their true propensity score",
                     "patients weighted by ps(1 - ps), ps the true propensity score"),
      stringsAsFactors = FALSE))
}

if (sys.nframe() == 0L) {
  cr_example <- make_cr_example(assign = ASSIGN, modifier = MODIFIER)
  usethis::use_data(cr_example, overwrite = TRUE)
}
```
